#### freebsd:func1.0
## EXSPECTATUM:
f () 
{ 
    failures=0;
    unset LC_ALL;
    export LC_CTYPE=en_US.ISO8859-1;
    i=128;
    set -f;
    while [ "$i" -le 255 ]; do
        c=$(printf \\"$(printf %o "$i")");
        ok=0;
        eval "$c() { ok=1; }";
        $c;
        ok1=$ok;
        ok=0;
        "$c";
        if [ "$ok" != 1 ] || [ "$ok1" != 1 ]; then
            echo "Bad results for character $i" 1>&2;
            : $((failures += 1));
        fi;
        unset -f $c;
        i=$((i+1));
    done;
    exit $((failures > 0))
}

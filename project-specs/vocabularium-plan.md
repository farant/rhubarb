**Goal:** A real vitrea app for capturing and drilling Latin vocabulary per book/chapter, built as a silicetum long-runner via `silex novum`, with every silex shortfall hand-carried and recorded on desideratum 01KZ6CSBQN.

/* vocabularium_pipa.c - vide caput */
#include "vocabularium_pipa.h"
#include "chorda_aedificator.h"
#include "xar.h"
#include <stdlib.h>

JsonValor*
voc_acta_legere (JsonValor* argumenta, Piscina* piscina,
    vacuum* datum, chorda* culpa)
{
    VocPipa*   pipa = (VocPipa*)datum;
    s64        post = ZEPHYRUM;
    Xar*       acta;
    JsonValor* tabulatum = json_tabulatum_creare(piscina);
    JsonValor* fructus   = json_objectum_creare(piscina);
    i32        i;

    (vacuum)culpa;
    si (argumenta != NIHIL)
    {
        JsonValor* v = json_objectum_capere(argumenta, "post_seq");

        si (v != NIHIL)
        {
            post = json_ad_integer(v);
        }
    }
    acta = volumen_acta_legere(pipa->volumen, post, piscina);
    si (acta != NIHIL)
    {
        per (i = ZEPHYRUM; i < xar_numerus(acta); i++)
        {
            VolumenActum* a = (VolumenActum*)xar_obtinere(acta, i);
            JsonValor*    o = json_objectum_creare(piscina);

            si (a == NIHIL)
            {
                perge;
            }
            json_objectum_ponere(o, "seq",
                json_integer_creare(piscina, a->seq));
            json_objectum_ponere(o, "momentum",
                json_chorda_creare(piscina, a->momentum));
            json_objectum_ponere(o, "genus",
                json_chorda_creare(piscina, a->genus));
            json_objectum_ponere(o, "datum",
                json_chorda_creare(piscina, a->datum));
            json_tabulatum_addere(tabulatum, o);
        }
    }
    json_objectum_ponere(fructus, "acta", tabulatum);
    redde fructus;
}

JsonValor*
voc_actum_appendere (JsonValor* argumenta, Piscina* piscina,
    vacuum* datum, chorda* culpa)
{
    VocPipa*   pipa = (VocPipa*)datum;
    JsonValor* genus_v;
    JsonValor* datum_v;
    chorda     textus;
    s64        seq;
    JsonValor* fructus;

    si (argumenta == NIHIL
        || (genus_v = json_objectum_capere(argumenta, "genus"))
            == NIHIL
        || (datum_v = json_objectum_capere(argumenta, "datum"))
            == NIHIL)
    {
        *culpa = chorda_ex_literis(
            "genus et datum requiruntur", piscina);
        redde NIHIL;
    }
    textus = json_scribere(datum_v, piscina);
    seq = volumen_actum_appendere(pipa->volumen,
        chorda_ut_cstr(json_ad_chorda(genus_v), piscina), textus);
    si (seq <= ZEPHYRUM)
    {
        constans character* err = volumen_error(pipa->volumen);

        *culpa = chorda_ex_literis((err != NIHIL)
            ? err : "volumen: actum non scriptum", piscina);
        redde NIHIL;
    }
    fructus = json_objectum_creare(piscina);
    json_objectum_ponere(fructus, "seq",
        json_integer_creare(piscina, seq));
    redde fructus;
}

constans character*
voc_viam_voluminis (Piscina* piscina)
{
    constans character* domus = getenv("HOME");
    ChordaAedificator*  aed;

    si (domus == NIHIL || domus[0] == '\0')
    {
        redde NIHIL;
    }
    aed = chorda_aedificator_creare(piscina, (memoriae_index)128);
    si (aed == NIHIL)
    {
        redde NIHIL;
    }
    chorda_aedificator_appendere_literis(aed, domus);
    chorda_aedificator_appendere_literis(aed,
        "/.rhubarb/vocabularium.volumen");
    redde chorda_ut_cstr(chorda_aedificator_finire(aed), piscina);
}

function vl_actum_applicare(st, a) {
  var d, v;
  try { d = JSON.parse(a.datum || 'null') || {}; }
  catch (e) { return; }
  if (a.genus === 'liber-conditus') {
    st.libri[a.seq] = { seq: a.seq, titulus: d.titulus || '' };
  } else if (a.genus === 'locus-positus') {
    st.locus = { liber: d.liber, capitulum: d.capitulum };
  } else if (a.genus === 'verbum-conditum') {
    st.verba[a.seq] = { seq: a.seq, liber: d.liber,
      capitulum: d.capitulum, lemma: d.lemma || '',
      formae: d.formae || '', genus: d.genus || '',
      significatio: d.significatio || '',
      exemplum: d.exemplum || '' };
  } else if (a.genus === 'verbum-emendatum') {
    v = st.verba[d.verbum];
    if (v) {
      v.lemma = d.lemma || ''; v.formae = d.formae || '';
      v.genus = d.genus || '';
      v.significatio = d.significatio || '';
      v.exemplum = d.exemplum || '';
    }
  } else if (a.genus === 'verbum-remotum') {
    delete st.verba[d.verbum];
    delete st.ultima[d.verbum];
  } else if (a.genus === 'responsum-datum') {
    if (st.verba[d.verbum]) {
      st.ultima[d.verbum] =
        { rectum: !!d.rectum, directio: d.directio };
    }
  }
  /* genera ignota (volumen-creatum, futura) innocue transeunt */
}

function vl_plicare(acta) {
  var st = { libri: {}, locus: null, verba: {}, ultima: {} };
  var i;
  for (i = 0; i < acta.length; i++) {
    vl_actum_applicare(st, acta[i]);
  }
  return st;
}

function vl_verba_ambitu(st, ambitus) {
  var claves = Object.keys(st.verba), exitus = [], i, v, u;
  for (i = 0; i < claves.length; i++) {
    v = st.verba[claves[i]];
    if (ambitus.genus === 'capitulum') {
      if (v.liber === ambitus.liber
          && v.capitulum === ambitus.capitulum) { exitus.push(v); }
    } else if (ambitus.genus === 'liber') {
      if (v.liber === ambitus.liber) { exitus.push(v); }
    } else if (ambitus.genus === 'omnia') {
      exitus.push(v);
    } else if (ambitus.genus === 'infirma') {
      u = st.ultima[v.seq];
      if (u && !u.rectum) { exitus.push(v); }
    }
  }
  exitus.sort(function (a, b) { return a.seq - b.seq; });
  return exitus;
}

function vl_capitula(st, liber) {
  var claves = Object.keys(st.verba), m = {}, exitus = [], i, v, k;
  for (i = 0; i < claves.length; i++) {
    v = st.verba[claves[i]];
    if (v.liber !== liber) { continue; }
    m[v.capitulum] = (m[v.capitulum] || 0) + 1;
  }
  for (k in m) {
    if (Object.prototype.hasOwnProperty.call(m, k)) {
      exitus.push({ capitulum: Number(k), numerus: m[k] });
    }
  }
  exitus.sort(function (a, b) { return a.capitulum - b.capitulum; });
  return exitus;
}

function vl_miscere(ordo, fortuita) {
  var i, j, t;
  for (i = ordo.length - 1; i > 0; i--) {
    j = Math.floor(fortuita() * (i + 1));
    t = ordo[i]; ordo[i] = ordo[j]; ordo[j] = t;
  }
  return ordo;
}

function vl_acervum_creare(verba, directio, fortuita) {
  return {
    ordo: vl_miscere(verba.slice(), fortuita),
    positio: 0, directio: directio,
    summa: verba.length, recta_prima: 0, iterationes: 0,
    visa: {}
  };
}

function vl_acervus_praesens(ac) {
  return (ac.positio < ac.ordo.length)
    ? ac.ordo[ac.positio] : null;
}

/* pravum -> charta in finem recyclatur; prima-vice semel notatur */
function vl_acervum_respondere(ac, rectum) {
  var v = vl_acervus_praesens(ac);
  if (!v) { return ac; }
  ac.iterationes++;
  if (!ac.visa[v.seq]) {
    ac.visa[v.seq] = true;
    if (rectum) { ac.recta_prima++; }
  }
  if (!rectum) { ac.ordo.push(v); }
  ac.positio++;
  return ac;
}

function vl_acervus_finis(ac) {
  return ac.positio >= ac.ordo.length;
}

function vl_frons(v, directio) {
  return (directio === 'la-en') ? v.lemma : v.significatio;
}
